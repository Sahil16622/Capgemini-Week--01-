import java.util.Scanner;

public class MaximumOfThreeNumbers {

    // Function to take an integer input from the user
    public static int takeInput(String prompt, Scanner scanner) {
        System.out.print(prompt);
        return scanner.nextInt();
    }

    // Function to calculate the maximum of three numbers
    public static int calculateMaximum(int num1, int num2, int num3) {
        int max = num1; // Assume num1 is the largest
        if (num2 > max) {
            max = num2; // Update max if num2 is larger
        }
        if (num3 > max) {
            max = num3; // Update max if num3 is larger
        }
        return max;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Taking input for three numbers
        int num1 = takeInput("Enter the first number: ", scanner);
        int num2 = takeInput("Enter the second number: ", scanner);
        int num3 = takeInput("Enter the third number: ", scanner);

        // Calculating the maximum of the three numbers
        int maximum = calculateMaximum(num1, num2, num3);

        // Displaying the result
        System.out.println("The maximum of the three numbers is: " + maximum);

        scanner.close();
    }
}
