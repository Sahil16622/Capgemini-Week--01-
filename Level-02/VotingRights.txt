import java.util.Random;
import java.util.Scanner;

public class VotingEligibility {

    // Method to generate random ages for students
    public static int[] generateAges(int numStudents) {
        Random random = new Random();
        int[] ages = new int[numStudents];

        for (int i = 0; i < numStudents; i++) {
            // Generate random 2-digit ages (10 to 99)
            ages[i] = random.nextInt(90) + 10;
        }

        return ages;
    }

    // Method to check voting eligibility and return 2D String array
    public static String[][] checkVotingEligibility(int[] ages) {
        String[][] result = new String[ages.length][2];

        for (int i = 0; i < ages.length; i++) {
            String age = String.valueOf(ages[i]);
            String canVote;

            // Validate and check voting eligibility
            if (ages[i] < 0) {
                canVote = "Invalid Age";
            } else if (ages[i] >= 18) {
                canVote = "Can Vote";
            } else {
                canVote = "Cannot Vote";
            }

            result[i][0] = age;
            result[i][1] = canVote;
        }

        return result;
    }

    // Method to display the 2D array in a tabular format
    public static void displayResults(String[][] data) {
        System.out.println("Age\tVoting Eligibility");
        System.out.println("-----------------------");

        for (String[] row : data) {
            System.out.println(row[0] + "\t" + row[1]);
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Take user input for the number of students
        System.out.print("Enter the number of students in the class: ");
        int numStudents = scanner.nextInt();

        // Generate random ages for students
        int[] studentAges = generateAges(numStudents);

        // Check voting eligibility
        String[][] votingResults = checkVotingEligibility(studentAges);

        // Display the results in tabular format
        System.out.println("\nVoting Eligibility of Students:");
        displayResults(votingResults);

        scanner.close();
    }
}
