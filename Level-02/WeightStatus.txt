import java.util.Scanner;

class Main {
    public static void main(String[] args) {

        // Importing Scanner class object to take input
        Scanner input = new Scanner(System.in);

        // Taking input for the number of persons to be evaluated
        System.out.print("Enter the number of persons to be evaluated: ");
        int noOfPerson = input.nextInt();

        // Creating arrays to track weight, height, BMI, and weight status
        int[] weight = new int[noOfPerson];
        int[] height = new int[noOfPerson];
        double[] bmi = new double[noOfPerson];
        String[] weightStatus = new String[noOfPerson];

        // Taking inputs for weight
        for (int i = 0; i < noOfPerson; i++) {
            System.out.print("Enter the weight (in kg) of person " + (i + 1) + ": ");
            weight[i] = input.nextInt();
        }

        // Taking inputs for height
        for (int i = 0; i < noOfPerson; i++) {
            System.out.print("Enter the height (in cm) of person " + (i + 1) + ": ");
            height[i] = input.nextInt();
        }

        // Calculating BMI and storing weight status
        for (int i = 0; i < noOfPerson; i++) {
            // Convert height to meters (from cm)
            double heightInMeters = height[i] / 100.0;

            // Calculate BMI using the formula BMI = weight / (height^2)
            bmi[i] = weight[i] / (heightInMeters * heightInMeters);

            // Determine weight status based on BMI
            if (bmi[i] <= 18.4) {
                weightStatus[i] = "Underweight";
            } else if (bmi[i] >= 18.5 && bmi[i] <= 24.9) {
                weightStatus[i] = "Normal";
            } else if (bmi[i] >= 25.0 && bmi[i] <= 39.9) {
                weightStatus[i] = "Overweight";
            } else {
                weightStatus[i] = "Obese";
            }
        }

        // Displaying the BMI and weight status of each individual
        System.out.println("\n--- Results ---");
        for (int i = 0; i < noOfPerson; i++) {
            System.out.println("Person " + (i + 1) + ":");
            System.out.println("  Weight: " + weight[i] + " kg");
            System.out.println("  Height: " + height[i] + " cm");
            System.out.println("  BMI: " + bmi[i]);
            System.out.println("  Weight Status: " + weightStatus[i]);
        }

        // Closing the scanner object
        input.close();
    }
}
