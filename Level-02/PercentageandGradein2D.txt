import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        
        //Taking input for number of students
        System.out.print("Enter the number of students to be examined : ");
        int noOfStudents = scanner.nextInt();
      

        //Creating the 2d array to store marks of all students in physics, chemistry and maths
        double [][] studentMarks = new double[noOfStudents][4];
        String [] grade = new String[noOfStudents];


        for(int i=0;i<noOfStudents;i++)
             {
                  //Entering the marks of student in physics
                  System.out.println("Enter the marks obtained by student "+(i+1)+" in physics");
                  studentMarks[i][0] = scanner.nextDouble();
             }


         for(int i=0;i<noOfStudents;i++)
             {
                  //Entering the marks of student in chemistry
                  System.out.println("Enter the marks obtained by student "+(i+1)+" in chemistry");
                  studentMarks[i][1] = scanner.nextDouble();
             }

          for(int i=0;i<noOfStudents;i++)
             {
                  //Entering the marks of student in maths
                  System.out.println("Enter the marks obtained by student "+(i+1)+" in maths");
                  studentMarks[i][2] = scanner.nextDouble();
             }



        // Calculate total marks, average, and percentage
        for(int i=0;i<noOfStudents;i++)
        {
        double totalMarks = studentMarks[i][0] + studentMarks[i][1] + studentMarks[i][2];
        double percentage = totalMarks / 3.0;
        studentMarks[i][3] = percentage;
        }

        for(int i=0;i<noOfStudents;i++)
        {
        // Determine the grade based on percentage
        if (studentMarks[i][3] >= 80) {
            System.out.println("Grade: A (Remarks:- Level-4, above agency-normalised standards.)");
        } else if (studentMarks[i][3] >= 70 && studentMarks[i][3] <=79) {
            System.out.println("Grade: B (Remarks:- Level-3, at agency-normalised standards.)");
        } else if (studentMarks[i][3] >= 60 && studentMarks[i][3] <=69) {
            System.out.println("Grade: C (Remarks:- Level-2, below but approaching agency-normalised standards.)");
        } else if (studentMarks[i][3] >= 50 && studentMarks[i][3] <=59) {
            System.out.println("Grade: D (Remarks:- Level-1, well below agency-normalised standards.)");
        } else if (studentMarks[i][3] >= 40 && studentMarks[i][3] <=49) {
            System.out.println("Grade: E (Remarks:- Level-(-1), too below agency-normalised standards.)");
        } else {
            System.out.println("Grade: R (Remarks:- Remedial standards)");
        }
        }
         scanner.close();
    }
}
