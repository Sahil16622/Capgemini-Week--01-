import java.util.Scanner;

public class DigitFrequency {
    public static void main(String[] args) {
        // Create a scanner object for user input
        Scanner input = new Scanner(System.in);

        // Take the input for the number
        System.out.print("Enter a number: ");
        int number = input.nextInt();

        // Define an array to store the frequency of digits (0-9)
        int[] frequency = new int[10];

        // Handle the number digit by digit
        while (number > 0) {
            int digit = number % 10; // Get the last digit
            frequency[digit]++;     // Increment the frequency of the digit
            number = number / 10;   // Remove the last digit
        }

        // Display the frequency of each digit
        System.out.println("Frequency of each digit:");
        for (int i = 0; i < 10; i++) {
            if (frequency[i] > 0) { // Display only digits that appear in the number
                System.out.println("Digit " + i + ": " + frequency[i]);
            }
        }

        // Close the scanner object
        input.close();
    }
}
