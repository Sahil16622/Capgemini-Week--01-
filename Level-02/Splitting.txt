import java.util.Scanner;

public class StringSplitter {

    // Method to calculate the length of a string without using length() method
    public static int calculateLength(String input) {
        int count = 0;
        try {
            while (true) {
                input.charAt(count); // Access each character
                count++;
            }
        } catch (IndexOutOfBoundsException e) {
            // Exception thrown when out-of-bounds is accessed
        }
        return count;
    }

    // Method to split the text into words without using split() method
    public static String[] customSplit(String text) {
        int length = calculateLength(text);
        int wordCount = 1; // At least one word exists

        // Count words based on spaces
        for (int i = 0; i < length; i++) {
            if (text.charAt(i) == ' ') {
                wordCount++;
            }
        }

        // Array to store indexes of spaces
        int[] spaceIndexes = new int[wordCount - 1];
        int index = 0;

        // Store space indexes
        for (int i = 0; i < length; i++) {
            if (text.charAt(i) == ' ') {
                spaceIndexes[index++] = i;
            }
        }

        // Array to store words
        String[] words = new String[wordCount];
        int start = 0;

        // Extract words using indexes
        for (int i = 0; i < wordCount - 1; i++) {
            int end = spaceIndexes[i];
            words[i] = text.substring(start, end);
            start = end + 1; // Move past the space
        }
        // Add the last word
        words[wordCount - 1] = text.substring(start, length);

        return words;
    }

    // Method to compare two string arrays
    public static boolean compareStringArrays(String[] array1, String[] array2) {
        if (array1.length != array2.length) {
            return false;
        }
        for (int i = 0; i < array1.length; i++) {
            if (!array1[i].equals(array2[i])) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Take user input
        System.out.print("Enter a text: ");
        String userInput = scanner.nextLine();

        // Split using the custom method
        String[] customSplitResult = customSplit(userInput);

        // Split using the built-in method
        String[] builtInSplitResult = userInput.split(" ");

        // Compare the two results
        boolean isEqual = compareStringArrays(customSplitResult, builtInSplitResult);

        // Display the results
        System.out.println("Custom split result: ");
        for (String word : customSplitResult) {
            System.out.println(word);
        }

        System.out.println("Built-in split result: ");
        for (String word : builtInSplitResult) {
            System.out.println(word);
        }

        System.out.println("Are the results equal? " + isEqual);

        scanner.close();
    }
}
