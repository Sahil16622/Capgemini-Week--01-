import java.util.Scanner;

public class TemperatureConversion {
    public static void main(String[] args) {
        // Create a Scanner object for user input
        Scanner input = new Scanner(System.in);

        // Declare the Fahrenheit variable
        double fahrenheit, celsiusResult;

        // Take user input for the temperature in Fahrenheit
        System.out.print("Enter the temperature in Fahrenheit: ");
        fahrenheit = input.nextDouble();

        // Convert Fahrenheit to Celsius using the formula
        celsiusResult = (fahrenheit - 32) * 5/9;

        // Print the result
        System.out.println(fahrenheit + " Fahrenheit is " + celsiusResult + " Celsius.");

        // Close the scanner
        input.close();
    }
}
