import java.util.Scanner;

public class UpperCaseConversion {

    public static void main(String[] args) {
        // Create a Scanner object for user input
        Scanner scanner = new Scanner(System.in);

        // Take user input for the text
        System.out.print("Enter the complete text: ");
        String input = scanner.nextLine();

        // Convert text to uppercase using the String built-in method
        String builtInUpperCase = input.toUpperCase();

        // Convert text to uppercase using the user-defined method
        String userDefinedUpperCase = convertToUpperCase(input);

        // Compare the two results using the user-defined comparison method
        boolean areEqual = compareStrings(builtInUpperCase, userDefinedUpperCase);

        // Display the results
        System.out.println("\nOriginal Text: " + input);
        System.out.println("Uppercase (Built-in Method): " + builtInUpperCase);
        System.out.println("Uppercase (User-defined Method): " + userDefinedUpperCase);
        System.out.println("Comparison Result: " + (areEqual ? "The results are the same!" : "The results are different."));

        // Close the scanner
        scanner.close();
    }

    // Method to convert text to uppercase using charAt()
    public static String convertToUpperCase(String text) {
        StringBuilder upperCaseText = new StringBuilder();

        for (int i = 0; i < text.length(); i++) {
            char ch = text.charAt(i);
            // Check if the character is lowercase
            if (ch >= 'a' && ch <= 'z') {
                // Convert to uppercase by subtracting 32 from ASCII value
                upperCaseText.append((char) (ch - 32));
            } else {
                // Append the character as it is
                upperCaseText.append(ch);
            }
        }

        return upperCaseText.toString();
    }

    // Method to compare two strings using charAt()
    public static boolean compareStrings(String str1, String str2) {
        // Check if the lengths are the same
        if (str1.length() != str2.length()) {
            return false;
        }

        // Compare each character
        for (int i = 0; i < str1.length(); i++) {
            if (str1.charAt(i) != str2.charAt(i)) {
                return false;
            }
        }

        return true;
    }
}
