import java.util.Scanner;

public class KilometerToMilesConverter {
    public static void main(String[] args) {
        // Create a Scanner object for user input
        Scanner input = new Scanner(System.in);

        // Declare a variable for kilometers
        double km;

        // Prompt the user for input
        System.out.print("Enter the distance in kilometers: ");
        km = input.nextDouble(); // Take input for kilometers

        // Conversion formula: 1 mile = 1.6 km
        double miles = km / 1.6;

        // Print the result
        System.out.println("The total miles is " + miles + " miles for the given " + km + " km.");
        
        // Close the scanner
        input.close();
    }
}
