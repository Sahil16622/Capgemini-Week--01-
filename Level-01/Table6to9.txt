import java.util.Scanner;

public class MultiplicationTable {
    public static void main(String[] args) {
        // Create a scanner object to take user input
        Scanner scanner = new Scanner(System.in);

        // Get user input for the number
        System.out.print("Enter a number to generate its multiplication table from 6 to 9: ");
        int number = scanner.nextInt();

        // Create an array to store the multiplication results
        int[] multiplicationTable = new int[4]; // We need results for 6, 7, 8, and 9

        // Calculate the multiplication results and store them in the array
        for (int i = 6; i <= 9; i++) {
            multiplicationTable[i - 6] = number * i;
        }

        // Display the multiplication table results
        for (int i = 6; i <= 9; i++) {
            System.out.println(number + " * " + i + " = " + multiplicationTable[i - 6]);
        }

        // Close the scanner
        scanner.close();
    }
}
