import java.util.Scanner;

public class EmployeeBonus {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        //Ask the user to provide the employee’s salary
        System.out.print("Enter the employee's salary: ");
        double salary = scanner.nextDouble();

        //Ask the user to input years of service
        System.out.print("Enter the years of service: ");
        int years = scanner.nextInt();



        //if the service provided is greater than 5 years then calculate bonus to be added
        if (years > 5) {
            double bonus = salary * 0.05;
            System.out.println("The bonus amount is: " + bonus);
        } 


          //If service provided by the user is less than five years then print that no bonus will be rewarded
          else {
            System.out.println("No bonus is awarded.");
        }
           scanner.close();
    }
}
