import java.util.Scanner;

public class CollinearPoints {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Input for 3 points A(x1, y1), B(x2, y2), C(x3, y3)
        System.out.print("Enter the coordinates for point A (x1, y1): ");
        int x1 = scanner.nextInt();
        int y1 = scanner.nextInt();

        System.out.print("Enter the coordinates for point B (x2, y2): ");
        int x2 = scanner.nextInt();
        int y2 = scanner.nextInt();

        System.out.print("Enter the coordinates for point C (x3, y3): ");
        int x3 = scanner.nextInt();
        int y3 = scanner.nextInt();

        // Check if points are collinear using the slope formula
        boolean collinearBySlope = areCollinearBySlope(x1, y1, x2, y2, x3, y3);
        if (collinearBySlope) {
            System.out.println("The points are collinear based on the slope formula.");
        } else {
            System.out.println("The points are NOT collinear based on the slope formula.");
        }

        // Check if points are collinear using the area of triangle formula
        boolean collinearByArea = areCollinearByArea(x1, y1, x2, y2, x3, y3);
        if (collinearByArea) {
            System.out.println("The points are collinear based on the area formula.");
        } else {
            System.out.println("The points are NOT collinear based on the area formula.");
        }

        scanner.close();
    }

    // Method to check if points are collinear using the slope formula
    public static boolean areCollinearBySlope(int x1, int y1, int x2, int y2, int x3, int y3) {
        // Calculate the slopes of AB and BC
        double slopeAB = (double) (y2 - y1) / (x2 - x1);
        double slopeBC = (double) (y3 - y2) / (x3 - x2);
        
        // Compare slopes to check if they are equal
        return slopeAB == slopeBC;
    }

    // Method to check if points are collinear using the area of triangle formula
    public static boolean areCollinearByArea(int x1, int y1, int x2, int y2, int x3, int y3) {
        // Calculate the area of the triangle
        double area = 0.5 * (x1 * (y2 - y3) + x2 * (y3 - y1) + x3 * (y1 - y2));
        
        // Points are collinear if the area is 0
        return area == 0;
    }
}
