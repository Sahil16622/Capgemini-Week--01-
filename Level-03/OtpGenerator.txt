public class OTPGenerator {

    public static void main(String[] args) {
        // Generate 10 OTPs and store them in an array
        int[] otps = new int[10];
        
        for (int i = 0; i < 10; i++) {
            otps[i] = generateOTP();
        }
        
        // Ensure that the OTPs are unique
        boolean isUnique = checkUnique(otps);
        
        if (isUnique) {
            System.out.println("All OTPs are unique.");
        } else {
            System.out.println("There are duplicate OTPs.");
        }
    }

    // Method to generate a 6-digit OTP number
    public static int generateOTP() {
        int otp = (int) (Math.random() * 900000) + 100000;  // Ensures a 6-digit number
        return otp;
    }

    // Method to ensure OTP numbers are unique
    public static boolean checkUnique(int[] otps) {
        // Compare each OTP with others to check for duplicates
        for (int i = 0; i < otps.length; i++) {
            for (int j = i + 1; j < otps.length; j++) {
                if (otps[i] == otps[j]) {
                    return false;  // If a duplicate is found, return false
                }
            }
        }
        return true;  // If no duplicates are found, return true
    }
}
