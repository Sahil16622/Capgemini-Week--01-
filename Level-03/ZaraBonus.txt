public class ZaraBonus {

    public static void main(String[] args) {
        int numEmployees = 10;
        
        // Create a 2D array to store the salary and years of service
        double[][] employees = new double[numEmployees][2];  // First column is salary, second column is years of service
        
        // Generate random salary and years of service for each employee
        generateEmployeeData(employees);
        
        // Calculate the new salary and bonus for each employee
        double[][] updatedSalaries = calculateNewSalaryAndBonus(employees);
        
        // Calculate the sum of old salaries, new salaries, and total bonus
        double[] totals = calculateTotalSalariesAndBonus(employees, updatedSalaries);
        
        // Display the results in a simple tabular format
        displayResults(employees, updatedSalaries, totals);
    }

    // Method to generate random salary and years of service for employees
    public static void generateEmployeeData(double[][] employees) {
        for (int i = 0; i < employees.length; i++) {
            // Generate a random salary between 30000 and 100000
            double salary = (Math.random() * 70000) + 30000;
            // Generate random years of service between 1 and 10
            int yearsOfService = (int) (Math.random() * 10) + 1;
            
            employees[i][0] = salary;  // First column is salary
            employees[i][1] = yearsOfService;  // Second column is years of service
        }
    }

    // Method to calculate the new salary and bonus based on years of service
    public static double[][] calculateNewSalaryAndBonus(double[][] employees) {
        double[][] updatedSalaries = new double[employees.length][3];  // [oldSalary, years, newSalary + bonus]
        
        for (int i = 0; i < employees.length; i++) {
            double salary = employees[i][0];
            int yearsOfService = (int) employees[i][1];
            double bonus = 0.0;
            
            // Apply bonus based on years of service
            if (yearsOfService > 5) {
                bonus = salary * 0.05;  // 5% bonus
            } else {
                bonus = salary * 0.02;  // 2% bonus
            }
            
            // New salary = old salary + bonus
            double newSalary = salary + bonus;
            
            // Store the old salary, years of service, and new salary + bonus in the updatedSalaries array
            updatedSalaries[i][0] = salary;  // Old Salary
            updatedSalaries[i][1] = yearsOfService;  // Years of service
            updatedSalaries[i][2] = newSalary;  // New salary (including bonus)
        }
        
        return updatedSalaries;
    }

    // Method to calculate the total old salary, total new salary, and total bonus
    public static double[] calculateTotalSalariesAndBonus(double[][] employees, double[][] updatedSalaries) {
        double oldSalarySum = 0.0;
        double newSalarySum = 0.0;
        double totalBonus = 0.0;
        
        for (int i = 0; i < employees.length; i++) {
            oldSalarySum += employees[i][0];  // Sum of old salaries
            newSalarySum += updatedSalaries[i][2];  // Sum of new salaries (including bonus)
            totalBonus += updatedSalaries[i][2] - employees[i][0];  // Total bonus = new salary - old salary
        }
        
        return new double[]{oldSalarySum, newSalarySum, totalBonus};
    }

    // Method to display the results in a simple tabular format
    public static void displayResults(double[][] employees, double[][] updatedSalaries, double[] totals) {
        System.out.println("Employee | Old Salary | Years of Service | New Salary | Bonus");
        System.out.println("------------------------------------------------------------");
        
        for (int i = 0; i < employees.length; i++) {
            System.out.println("Employee " + (i + 1) + " | " + employees[i][0] + " | " + (int) employees[i][1] + " | " + updatedSalaries[i][2] + " | " + (updatedSalaries[i][2] - employees[i][0]));
        }

        System.out.println("------------------------------------------------------------");
        System.out.println("Total | " + totals[0] + " | | " + totals[1] + " | " + totals[2]);
    }
}
