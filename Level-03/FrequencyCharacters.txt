import java.util.Scanner;

public class CharacterFrequency {

    // Method to find the frequency of characters in the string
    public static String[][] findFrequency(String text) {
        int[] freq = new int[256]; // Array to store frequency of characters (256 ASCII values)
        
        // Loop through the text to count the frequency of each character
        for (int i = 0; i < text.length(); i++) {
            char currentChar = text.charAt(i);
            freq[currentChar]++; // Increment the frequency of the character
        }

        // Create a 2D array to store characters and their frequencies
        String[][] result = new String[256][2];
        int resultIndex = 0;

        // Loop through the frequency array to find characters with non-zero frequencies
        for (int i = 0; i < 256; i++) {
            if (freq[i] > 0) {
                result[resultIndex][0] = String.valueOf((char) i); // Character
                result[resultIndex][1] = String.valueOf(freq[i]); // Frequency
                resultIndex++;
            }
        }

        // Create a new array with exact size for results and return
        String[][] finalResult = new String[resultIndex][2];
        System.arraycopy(result, 0, finalResult, 0, resultIndex);
        return finalResult;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Taking user input for the string
        System.out.print("Enter a string: ");
        String text = scanner.nextLine();

        // Find the frequency of characters
        String[][] frequency = findFrequency(text);

        // Display the frequency of each character
        System.out.println("Character | Frequency");
        for (int i = 0; i < frequency.length; i++) {
            System.out.println(frequency[i][0] + "         | " + frequency[i][1]);
        }
    }
}
